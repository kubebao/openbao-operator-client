package evidence

import (
	"slices"
	"testing"
)

func TestContractBaselineAttentionInventory(t *testing.T) {
	current, yes, one := true, true, int64(1)
	base := Observation{GenerationCurrent: &current, Available: &Condition{Type: "Available", Status: "True"}}
	for _, tc := range []struct {
		name, code string
		apply      func(*Observation)
	}{
		{"missing-generation", "generation_unknown", func(o *Observation) { o.GenerationCurrent = nil }},
		{"mismatched-generation", "generation_mismatch", func(o *Observation) { no := false; o.GenerationCurrent = &no }},
		{"missing-availability", "availability_unknown", func(o *Observation) { o.Available = nil }},
		{"paused", "paused", func(o *Observation) { o.Paused = &yes }},
		{"maintenance", "maintenance", func(o *Observation) { o.MaintenanceEnabled = &yes }},
		{"hold", "restore_hold", func(o *Observation) { o.RestoreHold = "request-1" }},
		{"break-glass", "break_glass", func(o *Observation) { o.BreakGlass = &BreakGlass{Active: &yes} }},
		{"backup", "backup_failures", func(o *Observation) { o.Backup = &Backup{ConsecutiveFailures: &one} }},
		{"stale-condition", "condition_generation_unconfirmed", func(o *Observation) { o.Conditions = []Condition{{Type: "Ready", Status: "True"}} }},
		{"adverse-condition", "reported_condition", func(o *Observation) {
			o.Conditions = []Condition{{Type: "OpenBaoSealed", Status: "True", GenerationCurrent: &current}}
		}},
		{"controller-error", "controller_error", func(o *Observation) { o.WorkloadError = &ControllerError{Reason: "Failed"} }},
		{"upgrade-error", "upgrade_failure", func(o *Observation) { o.Upgrade = &Upgrade{Failure: &ControllerError{Reason: "Failed"}} }},
	} {
		t.Run(tc.name, func(t *testing.T) {
			o := base
			tc.apply(&o)
			codes := []string{}
			for _, a := range ClusterAttention(&o) {
				codes = append(codes, a.Code)
			}
			if !slices.Equal(codes, []string{tc.code}) {
				t.Fatalf("attention = %v, want %s", codes, tc.code)
			}
		})
	}
}
