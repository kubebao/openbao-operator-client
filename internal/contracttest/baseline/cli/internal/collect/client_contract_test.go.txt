package collect

import (
	"encoding/json"
	"github.com/kubebao/openbao-operator-cli/internal/evidence"
	"k8s.io/apimachinery/pkg/apis/meta/v1/unstructured"
	"strings"
	"testing"
	"time"
)

func TestContractBaselineProjection(t *testing.T) {
	now := time.Date(2026, 11, 10, 12, 0, 0, 0, time.UTC)
	object := unstructured.Unstructured{Object: map[string]any{
		"metadata": map[string]any{"name": "bao", "namespace": "team", "uid": "cluster-1", "generation": int64(4)},
		"spec":     map[string]any{"replicas": int64(3)},
		"status": map[string]any{"observedGeneration": int64(4), "conditions": []any{
			map[string]any{"type": "Available", "status": "False", "observedGeneration": int64(3), "message": "UNSTRUCTURED_PROVIDER_CANARY"},
		}},
	}}
	got, err := ProjectV1Alpha1(&object, now)
	if err != nil {
		t.Fatal(err)
	}
	if got.Voters.Ready != nil || got.Paused != nil {
		t.Fatal("missing values became defaults")
	}
	if got.GenerationCurrent == nil || !*got.GenerationCurrent || got.Available.GenerationCurrent == nil || *got.Available.GenerationCurrent {
		t.Fatal("generation independence changed")
	}
	encoded, err := json.Marshal(got)
	if err != nil {
		t.Fatal(err)
	}
	if !strings.Contains(string(encoded), "UNSTRUCTURED_PROVIDER_CANARY") {
		t.Fatal("baseline no longer retains unstructured provider text")
	}
	conditions, _, _ := unstructured.NestedSlice(object.Object, "status", "conditions")
	conditions = append(conditions, conditions[0])
	if err := unstructured.SetNestedSlice(object.Object, conditions, "status", "conditions"); err != nil {
		t.Fatal(err)
	}
	if _, err := ProjectV1Alpha1(&object, now); err == nil {
		t.Fatal("baseline accepted duplicate conditions")
	}
}

func TestContractBaselineAssociation(t *testing.T) {
	object := unstructured.Unstructured{Object: map[string]any{
		"metadata": map[string]any{"uid": "request-1"}, "spec": map[string]any{"cluster": "bao"},
		"status": map[string]any{"target": map[string]any{"uid": "cluster-1"}, "execution": map[string]any{"targetUID": "other"}},
	}}
	snapshot := evidence.Snapshot{Target: evidence.Target{Name: "bao", UID: "cluster-1"}, Observation: &evidence.Observation{}}
	_, state := restoreOwner(&object, snapshot)
	if state != "conflict" {
		t.Fatalf("conflicting UIDs = %s", state)
	}
	delete(object.Object, "status")
	_, state = restoreOwner(&object, snapshot)
	if state != "unknown" {
		t.Fatalf("name-only association = %s", state)
	}
}
