// These assertions preserve historical behavior, including behavior deliberately corrected by the client contract.
func TestContractBaselineProjection(t *testing.T) {
	now := time.Date(2026, 11, 10, 12, 0, 0, 0, time.UTC)
	for _, tc := range []struct {
		name     string
		observed int64
		wantZero bool
	}{
		{"current", 4, true}, {"future-accepted", 5, true}, {"stale", 3, false},
	} {
		t.Run(tc.name, func(t *testing.T) {
			object := unstructured.Unstructured{Object: map[string]any{
				"metadata": map[string]any{"name": "bao", "namespace": "team", "uid": "cluster-1", "generation": int64(4)},
				"spec":     map[string]any{"replicas": int64(3)},
				"status": map[string]any{"observedGeneration": tc.observed, "conditions": []any{
					map[string]any{"type": "Available", "status": "False", "message": "UNSTRUCTURED_PROVIDER_CANARY"},
				}},
			}}
			got := Project(object, config.Connection{ID: "origin"}, config.Namespace{}, now)
			if (got.ReadyReplicas != nil) != tc.wantZero {
				t.Fatalf("readiness = %v", got.ReadyReplicas)
			}
			encoded, err := json.Marshal(got)
			if err != nil {
				t.Fatal(err)
			}
			if strings.Contains(string(encoded), "UNSTRUCTURED_PROVIDER_CANARY") {
				t.Fatal("provider message retained")
			}
			items, _, err := unstructured.NestedSlice(object.Object, "status", "conditions")
			if err != nil {
				t.Fatal(err)
			}
			items = append(items, items[0])
			if err := unstructured.SetNestedSlice(object.Object, items, "status", "conditions"); err != nil {
				t.Fatal(err)
			}
			if duplicate := Project(object, config.Connection{ID: "origin"}, config.Namespace{}, now); len(duplicate.Conditions) != 2 {
				t.Fatal("baseline no longer accepts duplicate conditions")
			}
		})
	}
}

func TestContractBaselineRecoveryAssociation(t *testing.T) {
	now := time.Date(2026, 11, 10, 12, 0, 0, 0, time.UTC)
	deployment := fleet.Deployment{ID: "deployment", Name: "bao", UID: "cluster-1"}
	makeRequest := func(name, uid string) unstructured.Unstructured {
		return unstructured.Unstructured{Object: map[string]any{
			"apiVersion": "openbao.org/v1alpha1", "kind": "OpenBaoRestore",
			"metadata": map[string]any{"name": name, "namespace": "team", "uid": name, "creationTimestamp": "2026-11-10T11:00:00Z"},
			"spec":     map[string]any{"cluster": "bao"},
			"status": map[string]any{"phase": "Failed", "target": map[string]any{"uid": "conflicting-uid"},
				"execution": map[string]any{"targetUID": uid}},
		}}
	}
	policy := fleet.Policy{StartupGrace: time.Minute}
	created := now.Add(-2 * time.Hour)
	for _, uid := range []string{"", "cluster-1"} {
		got := latestRecovery([]unstructured.Unstructured{makeRequest("restore-a", uid)}, deployment, created, now, policy)
		if got == nil || got.RequestName != "restore-a" {
			t.Fatalf("baseline association missing for UID %q", uid)
		}
	}
	a, b := makeRequest("restore-a", "cluster-1"), makeRequest("restore-b", "cluster-1")
	left := latestRecovery([]unstructured.Unstructured{a, b}, deployment, created, now, policy)
	right := latestRecovery([]unstructured.Unstructured{b, a}, deployment, created, now, policy)
	if left.RequestName != "restore-b" || right.RequestName != "restore-a" {
		t.Fatal("equal-time baseline changed")
	}
	deployment.Recovery = left
	evidence.Evaluate(&deployment, created, false, policy, now)
	if len(deployment.Issues) != 1 {
		t.Fatalf("failed recovery findings = %v", deployment.Issues)
	}
	// The baseline caller converts an optional denied read into no requests, then reevaluates.
	deployment.Recovery = latestRecovery(nil, deployment, created, now.Add(time.Minute), policy)
	evidence.Evaluate(&deployment, created, false, policy, now.Add(time.Minute))
	if len(deployment.Issues) != 0 {
		t.Fatal("baseline empty optional read no longer clears recovery")
	}
}
