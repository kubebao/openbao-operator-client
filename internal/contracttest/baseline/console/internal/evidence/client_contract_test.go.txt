package evidence

import (
	"github.com/kubebao/openbao-operator-console/internal/fleet"
	"testing"
	"time"
)

func TestContractBaselineRuleInventory(t *testing.T) {
	now := time.Date(2026, 11, 10, 12, 0, 0, 0, time.UTC)
	zero, three := int32(0), int32(3)
	old := now.Add(-48 * time.Hour)
	for _, tc := range []struct {
		name       string
		deployment fleet.Deployment
		rule       fleet.Rule
		severity   fleet.Severity
	}{
		{"readiness", fleet.Deployment{ReadyReplicas: &zero, DesiredReplicas: &three}, fleet.RuleReplicaReadiness, fleet.SeverityCritical},
		{"future-sealed", fleet.Deployment{Generation: 4, Conditions: []fleet.Condition{{Type: "OpenBaoSealed", Status: "True", ObservedGeneration: 5}}}, fleet.RuleSealed, fleet.SeverityCritical},
		{"upgrade", fleet.Deployment{UpgradeFailure: &fleet.OperationFailure{Reason: "Failed"}}, fleet.RuleUpgradeFailed, fleet.SeverityHigh},
		{"backup-failed", fleet.Deployment{Backup: &fleet.Backup{ConsecutiveFailures: 1}}, fleet.RuleBackupFailed, fleet.SeverityHigh},
		{"backup-never", fleet.Deployment{Backup: &fleet.Backup{}}, fleet.RuleBackupNeverSucceeded, fleet.SeverityHigh},
		{"backup-old", fleet.Deployment{Backup: &fleet.Backup{LastSuccessAt: &old}}, fleet.RuleBackupOverdue, fleet.SeverityHigh},
		{"restore-test", fleet.Deployment{Backup: &fleet.Backup{LastSuccessAt: &now, RestoreTest: &fleet.RestoreTest{Outcome: "failed"}}}, fleet.RuleRestoreTest, fleet.SeverityHigh},
		{"replication", fleet.Deployment{Replica: &fleet.Replica{HistoryConflict: true}}, fleet.RuleReplicaHistoryConflict, fleet.SeverityCritical},
		{"recovery", fleet.Deployment{Recovery: &fleet.Recovery{Phase: "failed"}}, fleet.RuleRecoveryInvestigate, fleet.SeverityHigh},
	} {
		t.Run(tc.name, func(t *testing.T) {
			d := tc.deployment
			d.ID = "deployment"
			Evaluate(&d, now.Add(-time.Hour), false, fleet.Policy{StartupGrace: time.Minute, MaxBackupAge: 24 * time.Hour}, now)
			if len(d.Issues) != 1 || d.Issues[0].Rule != tc.rule || d.Issues[0].Severity != tc.severity {
				t.Fatalf("issues = %+v", d.Issues)
			}
			if d.Issues[0].ID != "deployment:"+string(tc.rule) {
				t.Fatal("baseline issue identity changed")
			}
		})
	}
}
